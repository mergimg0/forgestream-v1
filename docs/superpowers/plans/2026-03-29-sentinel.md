# Sentinel Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Sentinel real-time cognitive amplification system — a daemon + mode terminals that monitor, analyze, and enhance Claude Code sessions with SOS-guaranteed GRPO self-improvement.

**Architecture:** A FastAPI daemon on localhost:9100 receives hook events from the working terminal, routes them to mode-specific queues, runs GRPO optimization every 3 turns, and delivers approved injections back via UserPromptSubmit hooks. Mode terminals (Claude Code instances) read their event queues, generate insights, and let the user curate/inject them. The dual-SOS pattern (weight-space GRPO in daemon + context-space pattern accumulation in terminals) guarantees convergence.

**Tech Stack:** Python 3.11+, FastAPI, uvicorn, pydantic (data models), pytest, httpx (test client). No LLM dependencies in daemon. Braintrust API for Layer 2 hydration.

**Spec:** `docs/superpowers/specs/2026-03-29-sentinel-design.md`

# ~/.claude/sentinel/models.py
"""Sentinel data models — Pydantic v2."""

from __future__ import annotations

from datetime import datetime
from typing import Optional
from uuid import uuid4

from pydantic import BaseModel, Field


class SentinelEvent(BaseModel):
    """Event received from working terminal hooks."""

    event_id: str = Field(default_factory=lambda: str(uuid4()))
    event_type: str = ""
    session_id: str = ""
    timestamp: datetime = Field(default_factory=datetime.utcnow)
    payload: dict = Field(default_factory=dict)


class Insight(BaseModel):
    """Insight generated by a mode terminal."""

    insight_id: str = ""
    mode: str = ""
    content: str = ""
    injectable_text: str = ""
    confidence: float = 0.0
    severity: str = "info"
    evidence: list[str] = Field(default_factory=list)
    suggested_action: str = "inject"
    status: str = "pending"
    created_at: datetime = Field(default_factory=datetime.utcnow)
    resolved_at: Optional[datetime] = None
    outcome: Optional[str] = None


class SOSState(BaseModel):
    """Sacred Object System state for a mode's GRPO optimization."""

    mode: str = ""
    weights: dict[str, float] = Field(default_factory=dict)
    orbit: list[float] = Field(default_factory=list)
    iteration: int = 0
    initial_weights: dict[str, float] = Field(default_factory=dict)
    convergence_gap: float = 1.0
    last_updated: datetime = Field(default_factory=datetime.utcnow)
    weight_bounds: dict[str, tuple[float, float]] = Field(default_factory=dict)
    noise_bounds: dict[str, tuple[float, float]] = Field(default_factory=dict)


class Interaction(BaseModel):
    """Record of a single turn for GRPO scoring."""

    turn: int = 0
    user_prompt: str = ""
    injections_delivered: list[str] = Field(default_factory=list)
    auto_injections: list[str] = Field(default_factory=list)
    response_quality: Optional[float] = None
    user_correction: bool = False
    task_completed: bool = False
    insights_adopted: list[str] = Field(default_factory=list)
    insights_dismissed: list[str] = Field(default_factory=list)
```

# ~/.claude/sentinel/state.py
"""State manager — file I/O, session context, event routing."""

from __future__ import annotations

import json
from datetime import datetime
from pathlib import Path
from typing import Optional

import config
from models import SentinelEvent, Insight, SOSState


class StateManager:
    """Manages all Sentinel state files."""

    def __init__(self, state_dir: Optional[Path] = None):
        self.state_dir = state_dir or config.STATE_DIR
        self._ensure_dirs()
        self._turn_count = 0
        self._files_touched: list[str] = []
        self._recent_prompts: list[dict] = []
        self._recent_tools: list[dict] = []

    def _ensure_dirs(self) -> None:
        self.state_dir.mkdir(parents=True, exist_ok=True)
        for mode in config.MODES:
            (self.state_dir / "modes" / mode).mkdir(parents=True, exist_ok=True)
        (self.state_dir / "grpo").mkdir(parents=True, exist_ok=True)

    # --- Session Signal (Layer 0) ---

    def update_signal(
        self,
        session_id: str,
        working_dir: str,
        current_file: str = "",
        turn_count: int = 0,
    ) -> None:
        signal = {
            "active": True,
            "session_id": session_id,
            "working_dir": working_dir,
            "current_file": current_file,
            "turn_count": turn_count,
            "last_event_ts": datetime.utcnow().isoformat() + "Z",
        }
        self._atomic_write(self.state_dir / "session_signal.json", signal)

    def read_signal(self) -> dict:
        path = self.state_dir / "session_signal.json"
        if not path.exists():
            return {"active": False}
        return json.loads(path.read_text())

    # --- Event Routing ---

    def route_event(self, event: SentinelEvent) -> None:
        """Write event to all mode queues."""
        line = event.model_dump_json() + "\n"
        for mode in config.MODES:
            events_file = self.state_dir / "modes" / mode / "events.jsonl"
            with events_file.open("a") as f:
                f.write(line)

    # --- Session Context (Layer 1) ---

    def update_context(self, event: SentinelEvent) -> None:
        """Update session context from an incoming event."""
        if event.event_type == "user_prompt":
            self._turn_count += 1
            prompt_text = event.payload.get("prompt", "")
            self._recent_prompts.append({
                "turn": self._turn_count,
                "summary": prompt_text[:200],
                "ts": event.timestamp.isoformat(),
            })
            # Keep last 20 prompts
            self._recent_prompts = self._recent_prompts[-20:]

        elif event.event_type == "post_tool":
            file_path = event.payload.get("file", "")
            if file_path and file_path not in self._files_touched:
                self._files_touched.append(file_path)
            tool_name = event.payload.get("tool_name", "")
            self._recent_tools.append({
                "tool": tool_name,
                "file": file_path,
                "turn": self._turn_count,
            })
            self._recent_tools = self._recent_tools[-30:]

        self._save_context(event.session_id)

    def _save_context(self, session_id: str) -> None:
        ctx = {
            "session_id": session_id,
            "turn_count": self._turn_count,
            "files_touched": self._files_touched,
            "recent_prompts": self._recent_prompts[-10:],
            "recent_tools": self._recent_tools[-10:],
            "updated_at": datetime.utcnow().isoformat() + "Z",
        }
        self._atomic_write(self.state_dir / "session_context.json", ctx)

    def read_context(self) -> dict:
        path = self.state_dir / "session_context.json"
        if not path.exists():
            return {"turn_count": 0, "files_touched": [], "recent_prompts": [], "recent_tools": []}
        return json.loads(path.read_text())

    # --- Injections ---

    def queue_injection(self, insight: Insight) -> None:
        """Append an approved insight to the pending injections file."""
        path = self.state_dir / "pending_injections.jsonl"
        entry = {
            "insight_id": insight.insight_id,
            "mode": insight.mode,
            "injectable_text": insight.injectable_text,
            "confidence": insight.confidence,
            "queued_at": datetime.utcnow().isoformat() + "Z",
        }
        with path.open("a") as f:
            f.write(json.dumps(entry) + "\n")

    def read_pending_injections(self) -> list[dict]:
        path = self.state_dir / "pending_injections.jsonl"
        if not path.exists():
            return []
        lines = path.read_text().strip().split("\n")
        return [json.loads(line) for line in lines if line.strip()]

    def clear_pending_injections(self) -> None:
        path = self.state_dir / "pending_injections.jsonl"
        if path.exists():
            path.write_text("")

    # --- SOS State ---

    def save_sos(self, state: SOSState) -> None:
        path = self.state_dir / "modes" / state.mode / "sos_state.json"
        self._atomic_write_str(path, state.model_dump_json(indent=2))

    def load_sos(self, mode: str) -> SOSState:
        path = self.state_dir / "modes" / mode / "sos_state.json"
        if path.exists():
            return SOSState.model_validate_json(path.read_text())
        # Return defaults

# ~/.claude/sentinel/grpo.py
"""GRPO engine with SOS axiom verification.

Ported from forgestream/governor/improvement.py WeightTuner.
Each mode is a Sacred Object System (Gashi, 2026):
- Monotone Improvement: E(δ(π)) ≥ E(π)
- Bounded Step: d(δ(π), π) ≤ Δ
- Constraint Preservation: C(π) ⟹ C(δ(π)))
Convergence follows as a free theorem (Theorem 2.5).
"""

from __future__ import annotations

import random
from typing import TYPE_CHECKING

import config

if TYPE_CHECKING:
    from models import Interaction
    from state import StateManager


def normalize_weights(weights: dict[str, float]) -> dict[str, float]:
    """Normalize weights to sum to 1.0, clamping to min 0.01."""
    clamped = {k: max(0.01, v) for k, v in weights.items()}
    total = sum(clamped.values())
    return {k: v / total for k, v in clamped.items()}


def generate_perturbations(
    weights: dict[str, float], n: int = 10, sigma: float = 0.05
) -> list[dict[str, float]]:
    """Generate N Gaussian-perturbed weight vectors, each normalized."""
    perturbations = []
    keys = list(weights.keys())
    for _ in range(n):
        perturbed = {
            k: max(0.01, weights[k] + random.gauss(0, sigma))
            for k in keys
        }
        total = sum(perturbed.values())
        perturbations.append({k: v / total for k, v in perturbed.items()})
    return perturbations


class SentinelGRPO:
    """GRPO engine for all Sentinel modes."""

    def __init__(self, state_manager: StateManager):
        self.state_manager = state_manager

    def tune(self, mode: str, interactions: list[Interaction]) -> dict[str, float]:
        """Run one GRPO cycle for a mode. Returns updated weights.

        SOS axioms are enforced:
        1. Monotone Improvement — reject if no improvement
        2. Bounded Step — 70/30 blend caps distance
        3. Constraint Preservation — all weights positive, sum to 1
        """
        sos = self.state_manager.load_sos(mode)
        current = sos.weights
        current_score = self._evaluate(mode, current, interactions)

        # Generate perturbations
        perturbations = generate_perturbations(
            current, n=config.GRPO_PERTURBATIONS, sigma=config.GRPO_SIGMA
        )

        # Score each perturbation
        scored = [
            (self._evaluate(mode, p, interactions), p) for p in perturbations
        ]
        scored.sort(key=lambda x: x[0], reverse=True)
        best_score, best_weights = scored[0]

        # SOS Axiom 1: Monotone Improvement gate
        if best_score <= current_score:
            sos.orbit.append(current_score)
            sos.iteration += 1
            sos.convergence_gap = 1.0 - current_score
            self.state_manager.save_sos(sos)
            self.state_manager.record_orbit(mode, current_score, current)
            return current

        # Blend 70/30 (SOS Axiom 2: Bounded Step)
        blend = config.GRPO_BLEND_RATIO
        blended = {
            k: (1 - blend) * current[k] + blend * best_weights[k]
            for k in current
        }
        normalized = normalize_weights(blended)

        # SOS Axiom 3: Constraint Preservation
        if not self._check_constraints(normalized):
            sos.orbit.append(current_score)
            sos.iteration += 1
            self.state_manager.save_sos(sos)
            self.state_manager.record_orbit(mode, current_score, current)
            return current

        # Accept update
        final_score = self._evaluate(mode, normalized, interactions)
        sos.weights = normalized
        sos.orbit.append(final_score)
        sos.iteration += 1
        sos.convergence_gap = 1.0 - final_score
        self.state_manager.save_sos(sos)
        self.state_manager.record_orbit(mode, final_score, normalized)
        return normalized

    def _evaluate(
        self, mode: str, weights: dict[str, float], interactions: list[Interaction]
    ) -> float:
        """Evaluate a weight vector against interaction history.

        Score = weighted combination of outcome signals, bounded in [0, 1].
        """
        if not interactions:
            return 0.0

        total_score = 0.0
        for interaction in interactions:
            turn_score = 0.0
            if interaction.task_completed:
                turn_score += 1.0
            if interaction.user_correction:
                turn_score -= 0.5
            if interaction.insights_adopted:
                turn_score += 0.3 * len(interaction.insights_adopted)
            if interaction.insights_dismissed:
                turn_score -= 0.1 * len(interaction.insights_dismissed)
            total_score += max(0.0, min(1.0, turn_score))

        raw = total_score / len(interactions)

        # Weight influence: higher weight on successful categories boosts score
        weight_bonus = sum(
            w * 0.1 for w in weights.values() if w > 0.15
        )
        return min(1.0, max(0.0, raw + weight_bonus * 0.1))

    def _check_constraints(self, weights: dict[str, float]) -> bool:
        """SOS Constraint predicate: weights positive, sum to 1."""
        if abs(sum(weights.values()) - 1.0) > 1e-6:
            return False
        return all(v > 0 for v in weights.values())

"""Fast-path auto-injection heuristics for Prompt Forge.

These run in the daemon at wire speed (no LLM).
Only fire at high confidence (≥ 0.9 by default).
"""

from __future__ import annotations

import re
from dataclasses import dataclass
from typing import Optional

import config


@dataclass
class AutoInjectResult:
    should_inject: bool
    injectable_text: str = ""
    confidence: float = 0.0
    category: str = ""
    reasoning: str = ""


def analyze_prompt(
    prompt: str,
    recent_files: list[str],
    strategy_weights: dict[str, float],
    confidence_threshold: float = config.AUTO_INJECT_CONFIDENCE_MIN,
) -> AutoInjectResult:
    """Analyze a user prompt for fast-path enhancement opportunities.

    Returns an AutoInjectResult. Only sets should_inject=True when
    confidence >= threshold AND the relevant strategy weight >= 0.2.
    """
    # Check if prompt already has file references
    has_file_refs = bool(re.search(r'\.\w{1,4}\b', prompt))  # e.g., "auth.py"

    # Rule 1: Short prompt with no file references → suggest recently edited files
    if (
        len(prompt) < 20
        and not has_file_refs
        and recent_files
        and strategy_weights.get("specificity", 0) >= 0.2
    ):
        files_str = ", ".join(recent_files[:3])
        return AutoInjectResult(
            should_inject=True,
            injectable_text=f"[SENTINEL:AUTO] Recently edited files: {files_str}",
            confidence=0.92,
            category="specificity",
            reasoning=f"Prompt is short ({len(prompt)} chars) with no file references",
        )

    # Rule 2: Prompt references no files but files were recently edited
    if (
        not has_file_refs
        and recent_files
        and len(prompt) >= 20
        and strategy_weights.get("context", 0) >= 0.2
    ):
        # Only if the prompt is medium-length (not already very specific)
        if len(prompt) < 80:
            files_str = ", ".join(recent_files[:3])
            return AutoInjectResult(
                should_inject=True,
                injectable_text=f"[SENTINEL:AUTO] Context: recently edited {files_str}",
                confidence=0.90,
                category="context",
                reasoning="Prompt has no file references but files were recently edited",
            )

    return AutoInjectResult(should_inject=False)

# --- Request/Response Models ---

class UserPromptRequest(BaseModel):
    prompt: str = ""
    session_id: str = ""
    cwd: str = ""
    timestamp: Optional[str] = None

class PreToolRequest(BaseModel):
    tool_name: str = ""
    tool_input: dict = {}
    session_id: str = ""

class PostToolRequest(BaseModel):
    tool_name: str = ""
    tool_input: dict = {}
    tool_output_summary: str = ""
    session_id: str = ""

class StopRequest(BaseModel):
    session_id: str = ""
    transcript_path: str = ""

class InjectRequest(BaseModel):
    insight_id: str = ""
    mode: str = ""
    injectable_text: str = ""
    confidence: float = 0.0

class HookResponse(BaseModel):
    additionalContext: str = ""

class InjectResponse(BaseModel):
    status: str = "queued"


# --- App Factory ---

def create_app(state_dir: Optional[Path] = None) -> FastAPI:
    state = StateManager(state_dir=state_dir)
    grpo = SentinelGRPO(state_manager=state)
    interactions: list = []  # Current session interactions

    @asynccontextmanager
    async def lifespan(app: FastAPI):
        # Startup: write PID
        pid_path = (state_dir or config.STATE_DIR) / "daemon.pid"
        pid_path.write_text(str(os.getpid()))
        yield
        # Shutdown: run final GRPO, remove PID
        for mode in config.MODES:
            if interactions:
                grpo.tune(mode, interactions)
        if pid_path.exists():
            pid_path.unlink()

    app = FastAPI(title="Sentinel Daemon", lifespan=lifespan)

    # --- Hook Endpoints ---

    @app.post("/hook/user_prompt", response_model=HookResponse)
    async def hook_user_prompt(req: UserPromptRequest) -> HookResponse:
        # Create event and route
        event = SentinelEvent(
            event_type="user_prompt",
            session_id=req.session_id,
            payload={"prompt": req.prompt, "cwd": req.cwd},
        )
        state.route_event(event)
        state.update_context(event)
        state.update_signal(
            session_id=req.session_id,
            working_dir=req.cwd,
            turn_count=state._turn_count,
        )

        # Track interaction
        from models import Interaction
        interactions.append(Interaction(
            turn=state._turn_count,
            user_prompt=req.prompt,
        ))

        # Check for GRPO trigger (every 3 turns)
        if state._turn_count > 0 and state._turn_count % config.GRPO_TRIGGER_TURNS == 0:
            for mode in config.MODES:
                grpo.tune(mode, interactions)

        # Check for auto-injection (Prompt Forge fast-path)
        ctx = state.read_context()
        auto = analyze_prompt(
            prompt=req.prompt,
            recent_files=ctx.get("files_touched", []),
            strategy_weights=state.load_sos("prompt_forge").weights,
        )

        # Collect all pending injections
        parts = []

        # Pending curated injections
        pending = state.read_pending_injections()
        if pending:
            for inj in pending[:config.MAX_CURATED_PER_TURN]:
                parts.append(inj["injectable_text"])
            state.clear_pending_injections()

        # Auto-injection (max 1 per turn)
        if auto.should_inject and len(parts) < config.MAX_CURATED_PER_TURN + config.MAX_AUTO_INJECT_PER_TURN:
            parts.append(auto.injectable_text)

        additional = "\n".join(parts)

        # Enforce token budget
        if len(additional) > config.MAX_TOKENS_PER_TURN * 4:  # rough char estimate
            additional = additional[:config.MAX_TOKENS_PER_TURN * 4]

        return HookResponse(additionalContext=additional)

    @app.post("/hook/pre_tool")
    async def hook_pre_tool(req: PreToolRequest) -> dict:
        event = SentinelEvent(
            event_type="pre_tool",
            session_id=req.session_id,
            payload={"tool_name": req.tool_name, "tool_input": req.tool_input},
        )
        state.route_event(event)
        return {}

    @app.post("/hook/post_tool")
    async def hook_post_tool(req: PostToolRequest) -> dict:
        event = SentinelEvent(
            event_type="post_tool",
            session_id=req.session_id,
            payload={
                "tool_name": req.tool_name,
                "file": req.tool_input.get("file_path", ""),
                "tool_output_summary": req.tool_output_summary,
            },
        )
        state.route_event(event)
        state.update_context(event)
        return {}

    @app.post("/hook/stop")
    async def hook_stop(req: StopRequest) -> dict:
        event = SentinelEvent(
            event_type="stop",
            session_id=req.session_id,
            payload={"transcript_path": req.transcript_path},
        )
        state.route_event(event)
        return {}

    # --- Injection Endpoint ---

    @app.post("/inject", response_model=InjectResponse)
    async def inject(req: InjectRequest) -> InjectResponse:
        insight = Insight(
            insight_id=req.insight_id,
            mode=req.mode,
            injectable_text=req.injectable_text,
            confidence=req.confidence,
            status="approved",
        )
        state.queue_injection(insight)
        return InjectResponse(status="queued")

    # --- Context Endpoints ---

    @app.get("/context/{mode}")
    async def get_context(mode: str) -> dict:
        ctx = state.read_context()
        sos = state.load_sos(mode)
        return {
            **ctx,
            "sos_state": {
                "weights": sos.weights,
                "orbit_position": sos.iteration,
                "last_evaluator": sos.orbit[-1] if sos.orbit else 0.0,
            },
        }

    # --- Context Hydration (Layer 2) ---

    @app.get("/context/{mode}/hydrate")
    async def hydrate_context(mode: str) -> dict:
        """Layer 2 hydration — return recent thinking blocks and detailed turns.

        Reads from Braintrust API if BRAINTRUST_API_KEY is set.
        Falls back to event queue summary if not available.
        """
        import os
        api_key = os.environ.get("BRAINTRUST_API_KEY", "")
        recent_turns = []

        if api_key:
            # Query Braintrust for recent trace data
            try:
                import httpx
                signal_data = state.read_signal()
                session_id = signal_data.get("session_id", "")
                if session_id:
                    # Query the Braintrust API for session spans
                    resp = httpx.get(
                        f"https://api.braintrust.dev/v1/project_logs/{state.load_sos(mode).mode}/fetch",
                        headers={"Authorization": f"Bearer {api_key}"},
                        params={"limit": 10},
                        timeout=5.0,
                    )
                    if resp.status_code == 200:
                        spans = resp.json().get("events", [])
                        for span in spans:
                            recent_turns.append({
                                "turn": span.get("metadata", {}).get("turn", 0),
                                "thinking_summary": span.get("metadata", {}).get("thinking", ""),
                                "tools_used": span.get("metadata", {}).get("tools", []),
                            })
            except Exception:
                pass  # Graceful degradation

        ctx = state.read_context()
        return {
            "recent_turns": recent_turns,
            "session_token_total": 0,  # Populated when Braintrust returns token data
            "session_context": ctx,
        }

    # --- Queue Endpoint ---

    @app.get("/queue/{mode}")
    async def get_queue(mode: str) -> dict:
        events_file = (state_dir or config.STATE_DIR) / "modes" / mode / "events.jsonl"
        event_count = 0
        if events_file.exists():
            event_count = len(events_file.read_text().strip().split("\n"))
        return {"mode": mode, "pending_events": event_count}

    # --- GRPO Endpoint ---

    @app.get("/grpo/{mode}")
    async def get_grpo(mode: str) -> dict:
        sos = state.load_sos(mode)
        return {
            "mode": mode,
            "weights": sos.weights,
            "orbit": sos.orbit,
            "iteration": sos.iteration,
            "convergence_gap": sos.convergence_gap,
        }

    # --- Health ---

    @app.get("/health")
    async def health() -> dict:
        return {
            "status": "ok",
            "modes": config.MODES,
            "turn_count": state._turn_count,
            "pid": os.getpid(),
        }

    return app


# --- CLI Entry Point ---

def main():
    import uvicorn

    app = create_app()

    def handle_sigterm(signum, frame):
        sys.exit(0)

    signal.signal(signal.SIGTERM, handle_sigterm)
    uvicorn.run(app, host=config.DAEMON_HOST, port=config.DAEMON_PORT)


if __name__ == "__main__":
    main()
```

"""End-to-end test: full injection lifecycle."""

import pytest
from httpx import AsyncClient, ASGITransport

import config
import daemon


@pytest.fixture
def tmp_daemon(tmp_path, monkeypatch):
    monkeypatch.setattr(config, "STATE_DIR", tmp_path)
    monkeypatch.setattr(config, "MODES_STATE_DIR", tmp_path / "modes")
    monkeypatch.setattr(config, "GRPO_DIR", tmp_path / "grpo")
    monkeypatch.setattr(config, "PENDING_INJECTIONS", tmp_path / "pending_injections.jsonl")
    monkeypatch.setattr(config, "SESSION_SIGNAL", tmp_path / "session_signal.json")
    monkeypatch.setattr(config, "SESSION_CONTEXT", tmp_path / "session_context.json")
    return daemon.create_app(state_dir=tmp_path)


@pytest.fixture
async def client(tmp_daemon):
    transport = ASGITransport(app=tmp_daemon)
    async with AsyncClient(transport=transport, base_url="http://test") as c:
        yield c


@pytest.mark.asyncio
class TestFullInjectionLifecycle:
    async def test_event_to_injection_to_delivery(self, client):
        """Simulate: user prompt → mode generates insight → user approves →
        injected on next prompt."""

        # 1. User sends a prompt in working terminal
        resp = await client.post("/hook/user_prompt", json={
            "prompt": "fix the bug",
            "session_id": "session-1",
            "cwd": "/projects/forgestream",
        })
        assert resp.status_code == 200
        # No injections yet
        assert resp.json().get("additionalContext", "") == ""

        # 2. Verify event was routed to mode queues
        for mode in config.MODES:
            queue = await client.get(f"/queue/{mode}")
            assert queue.json()["pending_events"] >= 1

        # 3. Mode terminal (Orange Team) generates and submits an insight
        resp = await client.post("/inject", json={
            "insight_id": "ot-2026-0329-001",
            "mode": "orange_team",
            "injectable_text": "[SENTINEL:ORANGE_TEAM] No input validation on user-supplied JSON",
            "confidence": 0.87,
        })
        assert resp.json()["status"] == "queued"

        # 4. User sends next prompt — should receive the injection
        resp = await client.post("/hook/user_prompt", json={
            "prompt": "now add the API endpoint",
            "session_id": "session-1",
            "cwd": "/projects/forgestream",
        })
        additional = resp.json().get("additionalContext", "")
        assert "[SENTINEL:ORANGE_TEAM]" in additional
        assert "input validation" in additional

        # 5. Injection should be cleared — next prompt gets nothing
        resp = await client.post("/hook/user_prompt", json={
            "prompt": "run the tests",
            "session_id": "session-1",
        })
        assert resp.json().get("additionalContext", "") == ""

    async def test_grpo_triggers_every_3_turns(self, client, tmp_path):
        """GRPO should run after every 3 user prompts."""
        for i in range(4):
            await client.post("/hook/user_prompt", json={
                "prompt": f"prompt {i}",
                "session_id": "session-1",
            })

        # After 3 turns, GRPO should have run — check orbit
        grpo_resp = await client.get("/grpo/prompt_forge")
        data = grpo_resp.json()
        assert data["iteration"] >= 1  # At least one GRPO cycle

    async def test_context_accumulates(self, client):
        """Session context should build up across multiple events."""
        await client.post("/hook/user_prompt", json={
            "prompt": "read the config file",
            "session_id": "s1", "cwd": "/project",
        })
        await client.post("/hook/post_tool", json={
            "tool_name": "Read", "tool_input": {"file_path": "config.py"},
            "tool_output_summary": "File read", "session_id": "s1",
        })
        await client.post("/hook/user_prompt", json={
            "prompt": "edit the auth module",
            "session_id": "s1",
        })

        ctx = await client.get("/context/prompt_forge")
        data = ctx.json()
        assert data["turn_count"] == 2
